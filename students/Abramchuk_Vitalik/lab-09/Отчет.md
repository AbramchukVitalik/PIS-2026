```python
syntax = "proto3";

package notes;

// ===== Service =====
service NotesService {
  rpc CreateNote (CreateNoteRequest) returns (CreateNoteResponse);
  rpc GetNote (GetNoteRequest) returns (NoteResponse);
  rpc StreamNotes (StreamNotesRequest) returns (stream NoteResponse);
}

// ===== Messages =====
message CreateNoteRequest {
  string notebook_id = 1;
  string author_id = 2;
  string content = 3;
}

message CreateNoteResponse {
  string note_id = 1;
}

message GetNoteRequest {
  string note_id = 1;
}

message StreamNotesRequest {
  string notebook_id = 1;
}

message NoteResponse {
  string note_id = 1;
  string notebook_id = 2;
  string author_id = 3;
  string content = 4;
}
```

### 2. gRPC Server

```python
from concurrent import futures
import grpc

from grpc_notes import notes_pb2, notes_pb2_grpc
from services.notes_service import NotesService

service = NotesService()


class NotesServicer(notes_pb2_grpc.NotesServiceServicer):

    def CreateNote(self, request, context):
        note = service.create_note(
            request.notebook_id,
            request.author_id,
            request.content
        )

        return notes_pb2.CreateNoteResponse(note_id=note["note_id"])

    def GetNote(self, request, context):
        note = service.get_note(request.note_id)

        if not note:
            context.set_code(grpc.StatusCode.NOT_FOUND)
            return notes_pb2.NoteResponse()

        return notes_pb2.NoteResponse(**note)

    def StreamNotes(self, request, context):
        for note in service.stream_notes(request.notebook_id):
            yield notes_pb2.NoteResponse(**note)


def serve():
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))

    notes_pb2_grpc.add_NotesServiceServicer_to_server(
        NotesServicer(),
        server
    )

    server.add_insecure_port("[::]:50051")
    server.start()
    server.wait_for_termination()


if __name__ == "__main__":
    serve()
```

### 3. gRPC Client

```python
import grpc
from grpc_notes import notes_pb2, notes_pb2_grpc


def run():
    channel = grpc.insecure_channel("localhost:50051")
    client = notes_pb2_grpc.NotesServiceStub(channel)

    # Create note
    response = client.CreateNote(
        notes_pb2.CreateNoteRequest(
            notebook_id="1",
            author_id="1",
            content="Hello gRPC"
        )
    )

    print("Created:", response.note_id)

    # Get note
    note = client.GetNote(
        notes_pb2.GetNoteRequest(note_id=response.note_id)
    )

    print("Fetched:", note)

    # Stream notes
    for n in client.StreamNotes(
        notes_pb2.StreamNotesRequest(notebook_id="1")
    ):
        print("STREAM:", n)


if __name__ == "__main__":
    run()
```

### 4. Бизнес-логика

```python
import uuid

class NotesService:
    def __init__(self):
        self.notes = {}

    def create_note(self, notebook_id, author_id, content):
        note_id = str(uuid.uuid4())

        note = {
            "note_id": note_id,
            "notebook_id": notebook_id,
            "author_id": author_id,
            "content": content
        }

        self.notes[note_id] = note
        return note

    def get_note(self, note_id):
        return self.notes.get(note_id)

    def stream_notes(self, notebook_id):
        for note in self.notes.values():
            if note["notebook_id"] == notebook_id:
                yield note
```

### 5. Тест gRPC

```python
import grpc
import threading

from grpc_notes import notes_pb2, notes_pb2_grpc
from grpc_notes.server import serve

    serve()


def test_create_and_get():

            author_id="1",
            content="Test"
        )
    )

    note = client.GetNote(
        notes_pb2.GetNoteRequest(note_id=created.note_id)
    )

    assert note.content == "Test"
```

## Контрольные вопросы

### 1. В чём преимущество gRPC над REST?

- бинарный протокол (Protobuf)
- HTTP/2 и мультиплексирование
- автоматическая генерация кода
- поддержка streaming

### 2. Почему Protocol Buffers быстрее JSON?

- бинарный формат
- меньше размер сообщений
- быстрее сериализация

### 3. Зачем нужен streaming в gRPC?

- передача данных в реальном времени
- уменьшение задержек
- экономия ресурсов
