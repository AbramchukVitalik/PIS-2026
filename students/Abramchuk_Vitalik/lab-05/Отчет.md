# 1. Доменная модель

```python
class Note:
    def __init__(self, note_id: str, notebook_id: str, author_id: str, content: str):
        self.note_id = note_id
        self.notebook_id = notebook_id
        self.author_id = author_id
        self.content = content
```

# 2. Command

```python
class CreateNoteCommand:
    def __init__(self, notebook_id: str, author_id: str, content: str):
        self.notebook_id = notebook_id
        self.author_id = author_id
        self.content = content
```

# 3. Repository (Infrastructure Adapter)

```python
class InMemoryNoteRepository:
    def __init__(self):
        self._storage = {}

    def save(self, note: Note):
        self._storage[note.note_id] = note
        return note

    def find_by_id(self, note_id: str):
        return self._storage.get(note_id)
```

# 4. Service Layer

```python
import uuid

class NoteService:
    def __init__(self, repository):
        self.repository = repository

    def create_note(self, command: CreateNoteCommand):
        note = Note(
            note_id=str(uuid.uuid4()),
            notebook_id=command.notebook_id,
            author_id=command.author_id,
            content=command.content
        )
        return self.repository.save(note)

    def get_note(self, note_id: str):
        return self.repository.find_by_id(note_id)
```

# 5. REST Controller (FastAPI)

```python
from fastapi import FastAPI

app = FastAPI()

repo = InMemoryNoteRepository()
service = NoteService(repo)

@app.post("/notes")
def create_note(request: dict):
    cmd = CreateNoteCommand(
        notebook_id=request["notebook_id"],
        author_id=request["author_id"],
        content=request["content"]
    )
    note = service.create_note(cmd)
    return {"id": note.note_id}

@app.get("/notes/{note_id}")
def get_note(note_id: str):
    note = service.get_note(note_id)
    return {
        "id": note.note_id,
        "notebook_id": note.notebook_id,
        "author_id": note.author_id,
        "content": note.content
    }
```

# 6. Интеграционные тесты

```python
from fastapi.testclient import TestClient
from infrastructure.adapter.inbound.note_controller import app

client = TestClient(app)

def test_create_note():
    response = client.post("/notes", json={
        "notebook_id": "1",
        "author_id": "1",
        "content": "Hello world"
    })

    assert response.status_code == 200
    assert "id" in response.json()


def test_get_note():
    create = client.post("/notes", json={
        "notebook_id": "1",
        "author_id": "1",
        "content": "Hello"
    })

    note_id = create.json()["id"]

    response = client.get(f"/notes/{note_id}")

    assert response.status_code == 200
    assert response.json()["id"] == note_id
```
